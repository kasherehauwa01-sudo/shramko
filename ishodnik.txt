<script>
const SK='shamko_v3';
function sw(n){
  document.querySelectorAll('.ws').forEach(s=>s.classList.remove('act'));
  document.querySelectorAll('.wt').forEach(t=>t.classList.remove('act'));
  document.getElementById('w'+n).classList.add('act');
  document.querySelector('.wt[data-w="'+n+'"]').classList.add('act');
}
function tog(id){
  const c=document.getElementById('day-'+id);
  c.classList.toggle('open');
}
function toggleMeal(btn,day,type){
  const meal=btn.closest('.meal');
  meal.classList.toggle('ck');
  const ck=meal.classList.contains('ck');
  btn.querySelector('.cc').textContent=ck?'✓':'';
  const card=meal.closest('.dc');
  const meals=card.querySelectorAll('.meal');
  const cked=card.querySelectorAll('.meal.ck');
  if(meals.length>0&&meals.length===cked.length){
    card.classList.add('all-done');
    if(day===20&&type==='dinner')setTimeout(()=>document.getElementById('celebration').classList.add('show'),400);
  }else{card.classList.remove('all-done');}
  save();upd();
}
function save(){
  const keys=[];
  document.querySelectorAll('.meal.ck').forEach(m=>{
    const day=m.closest('.dc').id;
    const type=m.classList.contains('breakfast')?'b':m.classList.contains('lunch')?'l':'d';
    keys.push(day+':'+type);
  });
  localStorage.setItem(SK,JSON.stringify(keys));
}
function load(){
  try{
    const saved=JSON.parse(localStorage.getItem(SK)||'[]');
    saved.forEach(k=>{
      const[dayId,type]=k.split(':');
      const card=document.getElementById(dayId);
      if(!card)return;
      const cls=type==='b'?'breakfast':type==='l'?'lunch':'dinner';
      const meal=card.querySelector('.meal.'+cls);
      if(!meal)return;
      meal.classList.add('ck');
      meal.querySelector('.cc').textContent='✓';
    });
    document.querySelectorAll('.dc').forEach(c=>{
      if(c.querySelectorAll('.meal').length===c.querySelectorAll('.meal.ck').length&&c.querySelectorAll('.meal').length>0)c.classList.add('all-done');
    });
    upd();
  }catch(e){}
}
function upd(){
  const done=document.querySelectorAll('.meal.ck').length;
  const pct=Math.round(done/60*100);
  document.getElementById('pf').style.width=pct+'%';
  document.getElementById('pl').textContent=done+' из 60 приёмов пищи';
}
document.addEventListener('DOMContentLoaded',()=>{load();document.getElementById('day-1').classList.add('open');});
</script>
